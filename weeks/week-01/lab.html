library(tidyverse)
library(NHANES)
theme_set(theme_minimal(base_size = 14))

nhanes_adults <- NHANES |>
  filter(SurveyYr == "2011_12", Age >= 20) |>
  distinct(ID, .keep_all = TRUE) |>
  select(
    id = ID, age = Age, gender = Gender, race = Race1,
    education = Education, income_mid = HHIncomeMid, poverty = Poverty,
    height = Height, weight = Weight, bmi = BMI,
    bp_sys = BPSysAve, bp_dia = BPDiaAve, pulse = Pulse,
    tot_chol = TotChol, sleep_hrs = SleepHrsNight,
    phys_active = PhysActive, smoker = Smoke100n,
    alcohol_days = AlcoholYear, diabetes = Diabetes,
    health_gen = HealthGen
  )
